    {
        protected Behavior(Bezoro.Behaviors.BehaviorCompositionLimits limits = default) { }

        protected void EndSignalOperation() { }

        protected abstract void OnStart(ref TContext context);

        protected abstract Bezoro.BehaviorTrees.BehaviorStatus OnTick(ref TContext context, ref Bezoro.BehaviorTrees.BehaviorBudget budget);

        protected void StartChild(Bezoro.Behaviors.Behavior<TContext> child, ref TContext context) { }
        public void Stop(ref TContext context) { }
        protected void StopChild(Bezoro.Behaviors.Behavior<TContext> child, ref TContext context) { }
        public Bezoro.BehaviorTrees.BehaviorStatus Tick(ref TContext context, ref Bezoro.BehaviorTrees.BehaviorBudget budget) { }
        protected Bezoro.BehaviorTrees.BehaviorStatus TickChild(Bezoro.Behaviors.Behavior<TContext> child, ref TContext context, ref Bezoro.BehaviorTrees.BehaviorBudget budget) { }
    }
