        Failure = 3,
        Faulted = 4,

    public sealed class BehaviorTreeRunner<TContext>
    {
        public BehaviorTreeRunner(Bezoro.BehaviorTrees.BehaviorTreeDefinition definition) { }
        public int ActiveNodeId { get; }
        public Bezoro.BehaviorTrees.BehaviorTreeDefinition Definition { get; }
        public int FrameCapacity { get; }
        public int ReactiveFrameCapacity { get; }

        public void Start() { }

        Bezoro.BehaviorTrees.BehaviorStatus Tick(int nodeId, int actionId, Bezoro.BehaviorTrees.BehaviorActivation activation, ref TContext context, ref Bezoro.BehaviorTrees.BehaviorBudget budget);
    }
